```python
import cv2
import numpy as np

class PerspectiveCorrector:
    def __init__(self):
        self.calibration_points = None
        self.transformation_matrix = None
        
    def calibrate_paper_bounds(self, frame):
        """Detect paper corners for perspective correction"""
        # Method 1: Manual corner selection
        corners = self.detect_paper_corners(frame)
        
        # Method 2: Automatic detection using paper edges
        # corners = self.auto_detect_paper_bounds(frame)
        
        # Define target rectangle (corrected view)
        target_width, target_height = 800, 600
        target_corners = np.float32([
            [0, 0], [target_width, 0], 
            [target_width, target_height], [0, target_height]
        ])
        
        # Calculate perspective transformation matrix
        self.transformation_matrix = cv2.getPerspectiveTransform(
            corners, target_corners
        )
    
    def correct_perspective(self, frame):
        """Apply real-time perspective correction"""
        if self.transformation_matrix is not None:
            corrected = cv2.warpPerspective(
                frame, self.transformation_matrix, (800, 600)
            )
            return corrected
        return frame
    
    def ai_observe_corrected_drawing(self, agent, raw_frame):
        """AI sees properly oriented drawing"""
        corrected_frame = self.correct_perspective(raw_frame)
        
        # AI analyzes its work without perspective distortion
        drawing_analysis = agent.analyze_drawing_progress(corrected_frame)
        return drawing_analysis
```

**2. Paper Detection Methods:**
```python
def detect_paper_corners(self, frame):
    """Detect white paper against background"""
    # Convert to HSV for better white detection
    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
    
    # Threshold for white paper
    lower_white = np.array([0, 0, 200])
    upper_white = np.array([255, 30, 255])
    mask = cv2.inRange(hsv, lower_white, upper_white)
    
    # Find contours and get largest rectangle
    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    
    if contours:
        largest_contour = max(contours, key=cv2.contourArea)
        # Approximate to rectangle
        epsilon = 0.02 * cv2.arcLength(largest_contour, True)
        approx = cv2.approxPolyDP(largest_contour, epsilon, True)
        
        if len(approx) == 4:
            return np.float32(approx.reshape(4, 2))
    
    return None

def auto_detect_paper_bounds(self, frame):
    """Alternative: Use edge detection for paper boundaries"""
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    edges = cv2.Canny(gray, 50, 150, apertureSize=3)
    
    # Hough line detection for paper edges
    lines = cv2.HoughLines(edges, 1, np.pi/180, threshold=100)
    
    # Find intersection points to determine corners
    corners = self.find_line_intersections(lines)
    return corners
```

```python
class CNCCalibrator:
    def __init__(self):
        self.calibration_grid = None
        self.distortion_map = None
    
    def create_test_grid(self, width=100, height=100, spacing=10):
        """Generate perfect test grid for CNC calibration"""
        grid_points = []
        for x in range(0, width, spacing):
            for y in range(0, height, spacing):
                grid_points.append([x, y])
        return np.array(grid_points)
    
    def analyze_drawn_grid(self, camera_frame):
        """Analyze what CNC actually drew vs. intended grid"""
        corrected_frame = self.perspective_corrector.correct_perspective(camera_frame)
        
        # Detect actual grid intersections
        actual_points = self.detect_grid_intersections(corrected_frame)
        intended_points = self.create_test_grid()
        
        # Calculate distortion mapping
        self.distortion_map = self.calculate_distortion_mapping(
            intended_points, actual_points
        )
    
    def correct_drawing_coordinates(self, drawing_coordinates):
        """Apply inverse distortion to compensate for CNC mechanics"""
        if self.distortion_map is not None:
            # Apply distortion correction to coordinates before G-code generation
            corrected_coords = self.apply_inverse_distortion(
                drawing_coordinates, self.distortion_map
            )
            return corrected_coords
        return drawing_coordinates
```
