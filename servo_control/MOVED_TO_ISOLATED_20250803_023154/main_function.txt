
def main():
    """Main function to start the clean cursor interface."""
    try:
        interface = CleanCursorInterface()
        interface.root.mainloop()
    except KeyboardInterrupt:
        print("\n👋 Shutting down...")
    except Exception as e:
        print(f"❌ Error: {e}")
        import traceback
        traceback.print_exc()

if __name__ == "__main__":
    main()
